cmake_minimum_required(VERSION 3.22)
project(singz-torrent LANGUAGES CXX)
find_package(LibtorrentRasterbar 2.1 CONFIG REQUIRED)
find_package(Threads REQUIRED)
# The SDK is always Release, including inside a Debug RN app. Libtorrent's
# exported Debug assertion macro changes public object layouts (bitfields).
# Match the linked SDK rather than inheriting the consumer configuration.
get_target_property(_lt_defs LibtorrentRasterbar::torrent-rasterbar INTERFACE_COMPILE_DEFINITIONS)
list(FILTER _lt_defs EXCLUDE REGEX "TORRENT_USE_ASSERTS")
set_property(TARGET LibtorrentRasterbar::torrent-rasterbar PROPERTY INTERFACE_COMPILE_DEFINITIONS "${_lt_defs}")
add_library(singz_torrent STATIC torrent_engine.cpp)
target_compile_features(singz_torrent PUBLIC cxx_std_17)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/torrent_engine.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/torrent_engine.cpp" SINGZ_TORRENT_SOURCE_STAMP)
target_compile_definitions(singz_torrent PRIVATE SINGZ_TORRENT_SOURCE_STAMP="${SINGZ_TORRENT_SOURCE_STAMP}")
set_target_properties(singz_torrent PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
 target_compile_definitions(singz_torrent PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_include_directories(singz_torrent PUBLIC .)
target_link_libraries(singz_torrent PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
install(TARGETS singz_torrent ARCHIVE DESTINATION lib)
install(FILES torrent_engine.h DESTINATION include)
if(SINGZ_TORRENT_TESTS)
 target_compile_definitions(singz_torrent PUBLIC SINGZ_TORRENT_TESTS=1)
 enable_testing()
 add_executable(torrent_core_tests torrent_core_tests.cpp)
 target_link_libraries(torrent_core_tests PRIVATE singz_torrent)
 add_test(NAME torrent_core_tests COMMAND torrent_core_tests)
endif()

if(SINGZ_TORRENT_NODE_INCLUDE)
 add_library(singz_torrent_addon MODULE node_addon.cpp)
 target_include_directories(singz_torrent_addon PRIVATE "${SINGZ_TORRENT_NODE_INCLUDE}")
 target_link_libraries(singz_torrent_addon PRIVATE singz_torrent)
 set_target_properties(singz_torrent_addon PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "singz-torrent")
 if(APPLE)
  target_link_options(singz_torrent_addon PRIVATE "-undefined" "dynamic_lookup")
 elseif(WIN32)
  target_link_libraries(singz_torrent_addon PRIVATE "${SINGZ_TORRENT_NODE_LIBRARY}")
 endif()
endif()

if(ANDROID)
 add_library(singztorrent SHARED android/torrent_jni.cpp)
 target_link_libraries(singztorrent PRIVATE singz_torrent)
 install(TARGETS singztorrent LIBRARY DESTINATION lib)
endif()
