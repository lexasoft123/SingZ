# Compile mutations of the implementation, never altered test expectations.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core.cpp" _core_original)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core.cpp")
foreach(control IN ITEMS floor clock missing gate diagnostics synthetic)
  if(control STREQUAL "floor")
    set(_old "std::ceil(2.0 * period * src / dst) + quantum")
    set(_new "std::ceil(2.0 * period * src / dst)")
    set(_expected "q==320")
  elseif(control STREQUAL "clock")
    set(_old "return rate;")
    set(_new "return nominalRate;")
    set(_expected "std::abs(rate-physical)<.1")
  elseif(control STREQUAL "missing")
    set(_old "if (active) s.missing += result.online ? std::min(count, result.missingFrames) : count;")
    set(_new "/* Deliberate mutant: omit missing-history accounting. */")
    set(_expected "renderer.missing==512")
  elseif(control STREQUAL "diagnostics")
    set(_old "row.flags|=IntervalRejected;")
    set(_new "{} /* Deliberate mutant: hide rejected clock intervals. */")
    set(_expected "(row.flags & ms::IntervalRejected)!=0")
  elseif(control STREQUAL "synthetic")
    set(_old "return origin + frameNs(frame, rate);")
    set(_new "return origin + frameNs(frame+1, rate);")
    set(_expected "ns==origin+ms::frameNs(frame,48000)")
  else()
    set(_old "if (!measured || !eligible || !std::isfinite(ms) || ms < 0) return Verdict::NotMeasured;")
    set(_new "if (!measured || !eligible || !std::isfinite(ms) || ms < 0) return Verdict::Pass;")
    set(_expected "unknown==ms::Verdict::NotMeasured")
  endif()
  string(REPLACE "${_old}" "${_new}" _mutated "${_core_original}")
  if(_mutated STREQUAL _core_original)
    message(FATAL_ERROR "Monitor spike ${control} mutation did not hit core.cpp")
  endif()
  set(_source "${CMAKE_CURRENT_BINARY_DIR}/core-negative-${control}.cpp")
  file(WRITE "${_source}" "${_mutated}")
  add_library(monitor_spike_core_negative_${control} STATIC "${_source}")
  target_compile_features(monitor_spike_core_negative_${control} PUBLIC cxx_std_20)
  target_include_directories(monitor_spike_core_negative_${control} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(monitor_spike_core_negative_${control} PUBLIC zCore::routing)
  singz_configure_realtime_target(monitor_spike_core_negative_${control})
  add_executable(monitor_spike_negative_${control} tests.cpp analysis.cpp
    "${ZCORE_ROOT}/zdsp/tests/allocation_trap.cpp")
  target_include_directories(monitor_spike_negative_${control} PRIVATE "${ZCORE_ROOT}/zdsp/tests")
  target_link_libraries(monitor_spike_negative_${control} PRIVATE monitor_spike_core_negative_${control} Threads::Threads)
  add_dependencies(monitor_spike_tests monitor_spike_negative_${control})
  add_test(NAME monitor_spike_negative_${control}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.py"
      "$<TARGET_FILE:monitor_spike_negative_${control}>" "${control}" "${_expected}")
  set_tests_properties(monitor_spike_negative_${control} PROPERTIES TIMEOUT 120)
endforeach()
